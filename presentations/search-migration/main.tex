\section{Problem \& Goal}
\subsection{Why and how this became an initiative}

\begin{frame}{Problem \& Goal: Initiative}
  \hypertarget{sec:problem}{}
  \begin{itemize}
    \item The search covers \textbf{multilingual} company names, with spelling and spacing variations.
    \item The current search worked and users \textbf{trusted} its \textbf{fuzzy matching} behavior.
    \item It ran on PostgreSQL with \textbf{16 CPU} cores and \textbf{64 GB} memory.
    \item At around \textbf{10M} records, \textbf{p99} latency was about \textbf{10 seconds}.
    \item Maximum sustainable \textbf{throughput} was around \textbf{2 requests per second}.
  \end{itemize}

  \vspace{0.5cm}
  \begin{block}{Goal}
    Bring \textbf{p99} search latency \textbf{under 1 second}, increase \textbf{throughput} by at least \textbf{10x}, and \textbf{preserve the ranking behavior} users already trusted.
  \end{block}
\end{frame}

\begin{frame}{Problem \& Goal: Options}
  After applying the available PostgreSQL and index optimizations, the remaining options were:
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.48\textwidth}
      \begin{itemize}
        \item Scale up the database.
          \begin{itemize}
            \item More CPU helped throughput and concurrency, but not single-query latency.
            \item It also kept search coupled to an expensive source-of-truth database.
          \end{itemize}
        \item Move search to a dedicated system.
      \end{itemize}

    \column{0.48\textwidth}
      \begin{block}{}

\section{Search Concepts}
\subsection{Text similarity vs search}

\begin{frame}{Search Concepts: Why This Matters}
  \hypertarget{sec:search-concepts}{}
  Because PostgreSQL \texttt{pg\_trgm} and Elasticsearch BM25 match and rank text differently, we need a few concepts before discussing the migration.

  \vspace{0.25cm}
  \begin{itemize}
    \item Trigram similarity: how the old search matched fuzzy text.
    \item Trigram indexes: why p99 latency still had a tail cost.
    \item BM25: how Elasticsearch scores analyzed search terms.
    \item Matching tradeoffs and query fuzziness: where behavior changes.
  \end{itemize}
\end{frame}

\begin{frame}{Search Concepts: Trigram Similarity}
  \begin{block}{What it is}
    A trigram is a rolling 3-character window. Similar strings share many windows, even when spelling or spacing differs.
  \end{block}

  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.48\textwidth}
      \begin{itemize}
        \item Useful for typos, spacing differences, and fuzzy names.
        \item Character overlap needs no language-specific stemmer.
        \item Normalization still matters for accents, case, and punctuation.
      \end{itemize}

    \column{0.44\textwidth}
      \begin{block}{Why it fits}
        \centering
        \large Names are short, noisy, and\\[0.12cm]
        users usually know only an approximate spelling.
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Search Concepts: Trigram Similarity Example}
  \begin{exampleblock}{PostgreSQL pads each word with spaces}
    \texttt{search} $\rightarrow$ \texttt{\{\_\_s, \_se, sea, ear, arc, rch, ch\_\}}\\[0.2cm]
    \texttt{serch} $\rightarrow$ \texttt{\{\_\_s, \_se, ser, erc, rch, ch\_\}}
  \end{exampleblock}
  \small \texttt{\_} represents a space; every fragment has three characters.
  \vspace{0.3cm}
  \begin{block}{Four shared fragments, nine distinct fragments}
    \centering \Large similarity = 4 / 9 $\approx$ 0.44
  \end{block}
  {\scriptsize\textcolor{MutedText}{Default \texttt{pg\_trgm.similarity\_threshold}: 0.3 for the \texttt{\%} filter; higher scores match, lower scores are excluded. Configurable per session.}}
\end{frame}

\begin{frame}{Search Concepts: Trigram Indexes}
  \begin{itemize}
    \item Common fragments can produce many candidate names.
    \item Lower thresholds admit more potential matches---and more candidates to process.
    \item Finding the top 20 may require scoring thousands of matches.
  \end{itemize}
  \begin{block}{Query plan matters}
    GIN filtering plus similarity sorting and GiST nearest-neighbor search have different costs. These latency results describe our workload, not a universal PostgreSQL limit.
  \end{block}
\end{frame}

\begin{frame}[fragile]{Search Concepts: Trigram Queries}
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.48\textwidth}
      \begin{block}{PostgreSQL: similarity}
        \scriptsize
\begin{verbatim}
SELECT name,
       similarity(name, 'serch')
FROM companies
WHERE name % 'serch'
ORDER BY
  similarity(name, 'serch') DESC
LIMIT 20;
\end{verbatim}
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Elasticsearch: analyzed terms}
        \scriptsize
\begin{verbatim}
GET companies/_search
{
  "size": 20,
  "query": {
    "match": {
      "name.ngram": "serch"
    }
  }
}
\end{verbatim}
      \end{block}
  \end{columns}
  \vspace{0.15cm}
  \small The Elasticsearch example analyzes both names and queries into trigrams. BM25 scores their shared terms; it does not reproduce PostgreSQL's similarity formula.
\end{frame}
\begin{frame}{Search Concepts: BM25 Formula}
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.44\textwidth}
      \begin{block}{Analyzer creates terms}
        \texttt{apple apple banana}\\[0.2cm]
        $\rightarrow$ \texttt{apple}, \texttt{apple}, \texttt{banana}
      \end{block}

      \begin{block}{Base idea}
        \centering
        $\text{score} \approx tf \times idf$
      \end{block}

      \begin{itemize}
        \item $tf$: term frequency in this document.
        \item $idf$: rarity across all documents.
      \end{itemize}

    \column{0.52\textwidth}
      \begin{block}{BM25 extension}
        \scriptsize
        \[
          \sum_{t \in q} \operatorname{idf}(t) \cdot
          \frac{tf(t,d)(k_1 + 1)}
          {tf(t,d) + k_1\left(1 - b + b\frac{|d|}{avgdl}\right)}
        \]
      \end{block}

      \begin{itemize}
        \item Rare terms still matter most.
        \item Repeated terms help, but the gain saturates.
        \item Long noisy documents are normalized against average length.
      \end{itemize}
  \end{columns}

  \vspace{0.05cm}
  \begin{block}{Key distinction}
    \texttt{pg\_trgm} is text similarity. BM25 scores analyzed search terms.
  \end{block}
\end{frame}

\begin{frame}{Search Concepts: Matching Tradeoffs}
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.48\textwidth}
      \begin{block}{Similarity wins}
        Query: \texttt{jon smyth}\\[0.15cm]
        Record: \texttt{john smith}
      \end{block}


    \column{0.48\textwidth}
      \begin{block}{BM25 wins}
        Query: \texttt{senior backend engineer}\\[0.15cm]
        Record: \texttt{backend engineer with search experience}
        \\[0.15cm]{\scriptsize With simple word / space tokenization}
      \end{block}

  \end{columns}
  \vspace{0.12cm}
  \begin{block}{Length-sensitive similarity: one- and two-character input}
    \begin{columns}[T,totalwidth=0.96\textwidth]
      \column{0.43\textwidth}
        \footnotesize
        \begin{tabular}{lcc}
          Candidate & Query \texttt{a} & Query \texttt{ab} \\
          \midrule
          \texttt{ab}    & \textcolor{WarmAccent}{0.25} & \textcolor{SecondaryTeal}{1.00} \\
          \texttt{abc}   & \textcolor{WarmAccent}{0.20} & \textcolor{SecondaryTeal}{0.40} \\
          \texttt{abcd}  & \textcolor{WarmAccent}{0.17} & \textcolor{SecondaryTeal}{0.33} \\
          \texttt{abcde} & \textcolor{WarmAccent}{0.14} & \textcolor{WarmAccent}{0.29} \\
        \end{tabular}
      \column{0.48\textwidth}
        \small Every query is a perfect prefix. Additional candidate trigrams dilute whole-string similarity.
        \par\smallskip
        \scriptsize\textcolor{SecondaryTeal}{Match: score $\geq 0.3$}\\
        \textcolor{WarmAccent}{No match: score $< 0.3$}
        \par\smallskip
        \textcolor{MutedText}{Controlled examples; scores rounded.}
    \end{columns}
  \end{block}
\end{frame}

\begin{frame}{Search Concepts: From Similarity to Relevance}
  \small
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.48\textwidth}
      \begin{block}{PostgreSQL: character overlap}
        How similar are these two names?
      \end{block}
      \begin{enumerate}
        \item Compare their trigram sets.
        \item Keep names above the similarity cutoff.
        \item Order by similarity.
      \end{enumerate}
      {\small\textcolor{MutedText}{Typos can still share many character fragments.}}
    \column{0.48\textwidth}
      \begin{block}{Elasticsearch: analyzed terms}
        Which matching names are most relevant?
      \end{block}
      \begin{enumerate}
        \item Analyze names and query text into terms.
        \item Match terms; optionally expand for typos.
        \item Rank text matches with BM25 and boosts.
      \end{enumerate}
      {\small\textcolor{MutedText}{BM25 scores matches; fuzziness handles term spelling differences.}}
  \end{columns}
  \vspace{0.2cm}
\end{frame}

\begin{frame}{Search Concepts: Query Shape and Fuzziness}
  \small
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.48\textwidth}
      \begin{block}{Boolean structure}
        Six scoring clauses under \texttt{bool.should}.
      \end{block}
      \begin{itemize}
        \item \texttt{minimum\_should\_match: 1}.
        \item No \texttt{must} or \texttt{filter} clause.
        \item Scores add when multiple clauses match.
      \end{itemize}
    \column{0.48\textwidth}
      \begin{block}{Two fuzzy clauses}
        Fuzziness is applied only where needed.
      \end{block}
      \begin{itemize}
        \item \texttt{name}: \texttt{AUTO} fuzziness.
        \item \texttt{name.prefix}: fuzziness \texttt{1}.
        \item No explicit \texttt{prefix\_length} or \texttt{max\_expansions} override.
      \end{itemize}
  \end{columns}
  \vspace{0.15cm}
  \begin{block}{Four non-fuzzy signals}
    Normalized exact term, standard prefix, whitespace-tokenized text, and word shingles.
  \end{block}
  {\scriptsize\textcolor{MutedText}{Clause weights are ranking fine-tuning, so they are intentionally omitted here.}}
\end{frame}




\begin{frame}{Search Concepts: Query Shape}
  \begin{block}{Our query combines six \texttt{should} clauses}
    Exact name, prefix, fuzzy prefix, fuzzy text, whitespace-tokenized text, and word shingles.
  \end{block}
  \begin{itemize}
    \item \texttt{minimum\_should\_match: 1} requires one matching clause.
    \item Matching clauses contribute to the combined score.
    \item Boosts weight signals; they do not guarantee a fixed ranking order.
  \end{itemize}
  \small The complete query and its boosts follow in Index Design.
\end{frame}

\section{Migration Design}
\subsection{Moving data without overloading PostgreSQL}

\begin{frame}{Migration Design: Requirements}
  \hypertarget{sec:migration-design}{}
  \begin{itemize}
    \item Deploy Elasticsearch quickly with low operational overhead.
    \item Replicate required PostgreSQL data into Elasticsearch.
    \item Preserve fuzzy, language-agnostic name search behavior.
    \item Benchmark old and new ranking behavior.
    \item Prove the new path before production rollout.
  \end{itemize}
\end{frame}

\begin{frame}{Migration Design: Before And After}
  \begin{block}{Before}
    Application $\rightarrow$ PostgreSQL\\[0.15cm]
    Transactional reads/writes and \texttt{pg\_trgm} search share the database.
  \end{block}
  \vspace{0.15cm}
  \begin{block}{After}
    Application $\rightarrow$ PostgreSQL for transactional data\\[0.15cm]
    Application $\rightarrow$ Elasticsearch for search\\[0.15cm]
    PostgreSQL $\rightarrow$ CDC pipeline $\rightarrow$ Elasticsearch
  \end{block}
  \small The migration changes search queries and adds asynchronous indexing.
\end{frame}

\begin{frame}{Migration Design: Elasticsearch Deployment}
  \begin{itemize}
    \item Deployment: managed by Elastic(Cloud Hosted).
    \item Default country index: \textbf{3 primary shards}.
    \item Replicas per primary: \textbf{1}
    \item Refresh interval: \textbf{300 seconds (5 minutes)}.
    \item Elasticsearch handles shard allocation, routing, and replication.
  \end{itemize}
  \begin{block}{Why Cloud Hosted}
    We chose Elastic Cloud Hosted for greater control over index settings, shards, and replicas.
  \end{block}
\end{frame}

\begin{frame}{Migration Design: Document Identity}
  \begin{itemize}
    \item Reuse the PostgreSQL row ID as the Elasticsearch document ID.
    \item Trace the same entity across the database, CDC, and search.
    \item Repeating a document replacement avoids duplicate documents.
  \end{itemize}
  \begin{block}{Replay also needs ordering}
    A stable ID alone cannot stop an older change overwriting a newer one. Preserve source order or enforce source versions, including deletes.
  \end{block}
\end{frame}

\begin{frame}{Migration Design: Checkpointing}
  \begin{block}{One partition's ordered batch}
    Read checkpoint $\rightarrow$ fetch rows $\rightarrow$ bulk write\\[0.2cm]

  \end{block}

\section{Index Design}
\subsection{Recreating fuzzy-name signals in Elasticsearch}

\begin{frame}{Index Design: Country-Specific Indexes}
  \hypertarget{sec:index-design}{}
  \begin{itemize}
    \item We maintain a separate search index for each indexed country.
    \item Languages and search settings differ between countries.
    \item The overall multi-field analyzer approach stays similar.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Index Design: Elasticsearch Multi-fields}
  \scriptsize
\begin{verbatim}
"name": {
  "type": "text", "analyzer": "language_text_standard",
  "fields": {
    "normalized_keyword": {
      "type": "keyword", "normalizer": "language_normalizer"
    },
    "prefix": {
      "type": "text", "analyzer": "prefix_analyzer",
      "search_analyzer": "keyword_normalized"
    },
    "whitespace_tokenizer": {
      "type": "text", "analyzer": "language_text_whitespace"
    },
    "word_shingle": {
      "type": "text", "analyzer": "word_shingle_analyzer"
    }
  }
}
\end{verbatim}
  \small Excerpt from the supplied default mapping; other subfields omitted.
\end{frame}

\begin{frame}{Index Design: Analyzer Choices}
  \small
  \begin{itemize}
    \item Normalize case and accents with \href{https://www.elastic.co/docs/reference/text-analysis/analysis-lowercase-tokenfilter}{\textcolor{PrimaryBlue}{\underline{lowercase}}} + \href{https://www.elastic.co/docs/reference/text-analysis/analysis-asciifolding-tokenfilter}{\textcolor{PrimaryBlue}{\underline{ASCII folding}}}.
    \item Main text: \href{https://www.elastic.co/docs/reference/text-analysis/analysis-standard-tokenizer}{\textcolor{PrimaryBlue}{\underline{standard tokenizer}}} with country-specific analysis and \href{https://www.elastic.co/docs/reference/text-analysis/analysis-stemmer-tokenfilter}{\textcolor{PrimaryBlue}{\underline{stemming}}}.
    \item Prefix: \href{https://www.elastic.co/docs/reference/text-analysis/analysis-keyword-tokenizer}{\textcolor{PrimaryBlue}{\underline{keyword tokenizer}}}, \href{https://www.elastic.co/docs/reference/text-analysis/analysis-edgengram-tokenfilter}{\textcolor{PrimaryBlue}{\underline{edge n-grams}}} 3--20; keyword query analysis.
    \item \href{https://www.elastic.co/docs/reference/text-analysis/analysis-whitespace-tokenizer}{\textcolor{PrimaryBlue}{\underline{Whitespace}}}: separate word boundaries and a \href{https://www.elastic.co/docs/reference/text-analysis/analysis-keyword-marker-tokenfilter}{\textcolor{PrimaryBlue}{\underline{legal-suffix protection filter}}}.
    \item \href{https://www.elastic.co/docs/reference/text-analysis/analysis-shingle-tokenfilter}{\textcolor{PrimaryBlue}{\underline{Shingles}}}: whitespace tokens, 2--3 words, plus individual words.
  \end{itemize}
  \begin{block}{Why custom prefix rather than search-as-you-type?}
    Our custom analyzer keeps whole-name prefixes, preserving \texttt{a-m} versus \texttt{a m}, with explicit 3--20-character limits and separate query boosts.
    \par\smallskip
    \href{https://www.elastic.co/docs/reference/elasticsearch/mapping-reference/search-as-you-type}{\underline{\texttt{search\_as\_you\_type}}} builds shingle/prefix subfields automatically and also supports custom analysis; we chose explicit control over these representations.
  \end{block}
  {\scriptsize\textcolor{MutedText}{Underlined components link to Elastic documentation; analyzer names and their combination are ours.}}
\end{frame}

\begin{frame}{Index Design: Prefix Matching}
  \begin{exampleblock}{Input}
    \texttt{progema}
  \end{exampleblock}

  \vspace{0.05cm}
  \begin{center}
    \texttt{pro}, \texttt{prog}, \texttt{proge}, \texttt{progem}, \texttt{progema}
  \end{center}

  \vspace{0.45cm}
  \begin{block}{Use case}
    Prefixes cover the start of the whole name (3--20 characters), including spaces and hyphens. The query stays one normalized token; 1--2-character prefixes are not indexed in this field.
  \end{block}
\end{frame}

\begin{frame}{Index Design: Why Separate Prefixes}
  \begin{itemize}
    \item Analyzer behavior on names such as \texttt{a-m} and \texttt{a m} drove the design.
    \item Separate fields retain different token boundaries and matching signals.
    \item Prefix matching has its own weighting, with and without fuzziness.
  \end{itemize}
  \begin{block}{Why not \texttt{search\_as\_you\_type}?}
    We needed control over these representations. That field type also accepts custom analysis; this example explains our multi-field choice, not a universal limitation of the alternative.
  \end{block}
\end{frame}

\begin{frame}{Index Design: Punctuation Changes Tokens}
  \begin{center}
    \begin{tabular}{lll}
      \toprule
      Tokenizer & Input: \texttt{a-m} & Input: \texttt{a m} \\
      \midrule
      Standard & \texttt{[a, m]} & \texttt{[a, m]} \\
      Whitespace & \texttt{[a-m]} & \texttt{[a, m]} \\
      Keyword & \texttt{[a-m]} & \texttt{[a m]} \\
      \bottomrule
    \end{tabular}
  \end{center}
  \begin{block}{Preserve the useful distinction}
    Standard tokenization merges these two inputs into the same terms. Whitespace and keyword representations retain their different forms.
  \end{block}
  \small Tokenizer-only examples; production char filters and token filters can change the result. Verify the full chain with \texttt{\_analyze}.
\end{frame}

\begin{frame}{Index Design: Character N-Grams}
  \begin{exampleblock}{Input}
    \texttt{progema}
  \end{exampleblock}

  \vspace{0.05cm}
  \begin{center}
    \texttt{pro}, \texttt{rog}, \texttt{oge}, \texttt{gem}, \texttt{ema}
  \end{center}

  \vspace{0.45cm}
  \begin{block}{Use case}
    \texttt{name.character\_ngram} is mapped with keyword tokenization and 3-grams. It is not used by the supplied six-clause query.
  \end{block}
\end{frame}

\begin{frame}{Index Design: Word Shingles}
  \begin{exampleblock}{Input: \texttt{progema fastighets ab}}
    Two-word shingles:\\[0.15cm]
    \texttt{progema fastighets}, \texttt{fastighets ab}\\[0.2cm]
    Three-word shingle:\\[0.15cm]
    \texttt{progema fastighets ab}
  \end{exampleblock}
  \begin{itemize}
    \item Shingles join adjacent tokens in their original order.
    \item They add phrase signals alongside individual words.
    \item They are a text-analysis technique, unrelated to \texttt{pgvector}.
  \end{itemize}
\end{frame}

\begin{frame}{Index Design: Size And Query Cost}
  \begin{center}
    \begin{tabular}{ll}
      \toprule
      Indexed form & Additional work \\
      \midrule
      Exact keyword & One normalized term per name \\
      Prefix & 3--20-character prefixes of the whole name \\
      Whitespace text & Another token stream \\
      Word shingle & Additional adjacent-word terms \\
      \bottomrule
    \end{tabular}
  \end{center}
  \begin{block}{Measure the trade-off}
    Index a representative sample. Compare disk usage, indexing throughput, query latency, and relevance before retaining an extra field.
  \end{block}
\end{frame}

\begin{frame}[fragile]{Index Design: Our Search Query}
  \scriptsize
\begin{verbatim}
body = {"query": {"bool": {
    "should": [
        {"term": {"name.normalized_keyword":
            {"value": query, "boost": 300}}},
        {"match": {"name.prefix":
            {"query": query, "boost": 40}}},
        {"match": {"name.prefix":
            {"query": query, "boost": 40, "fuzziness": "1"}}},
        {"match": {"name":
            {"query": query, "boost": 70, "fuzziness": "auto"}}},
        {"match": {"name.whitespace_tokenizer":
            {"query": query, "boost": 30}}},
        {"match": {"name.word_shingle":
            {"query": query, "boost": 10}}},
    ],
    "minimum_should_match": 1,
}}}
\end{verbatim}
  \small Python request body; \texttt{query} is the same input string in every clause.
\end{frame}

\begin{frame}{Index Design: Query Weights}
  \small
  \begin{center}
    \begin{tabular}{lll}
      \toprule
      Field & Match behavior & Boost \\
      \midrule
      \texttt{name.normalized\_keyword} & Exact term & 300 \\
      \texttt{name.prefix} & Analyzed prefix & 40 \\
      \texttt{name.prefix} & Fuzziness 1 & 40 \\
      \texttt{name} & Fuzziness auto & 70 \\
      \texttt{name.whitespace\_tokenizer} & Analyzed text & 30 \\
      \texttt{name.word\_shingle} & Adjacent-word signal & 10 \\
      \bottomrule
    \end{tabular}
  \end{center}
  \begin{block}{How the signals combine}
    At least one clause must match. Scores add across matching clauses; both prefix clauses can contribute to the same result.
  \end{block}
\end{frame}

\subsection{Proving relevance before production rollout}

\begin{frame}{Validation: Generated Queries with NLPAUG}
  \hypertarget{sec:validation}{}
  Use \href{https://github.com/makcedward/nlpaug}{\underline{NLPAUG}} and search-specific transformations to generate:
  \begin{itemize}
    \item Keyboard-adjacent and OCR-like substitutions.
    \item Character insertion, substitution, swapping, and deletion.
    \item Word swapping, cropping, deletion, and splitting.
    \item Spacing and casing variations.
    \item Keyword slicing and query truncation: partial names or dropped trailing words.
  \end{itemize}
  \vspace{0.1cm}
  \small Compare every generated query across both search setups; then validate historical queries through expert review of both top-20 lists.
  \par\smallskip
  {\scriptsize\textcolor{MutedText}{NLPAUG provides the noise augmenters; keyword slicing/query truncation is our search-specific test generation.}}
\end{frame}

\begin{frame}{Validation: What MRR Measures}
  Mean Reciprocal Rank: average {\boldmath$\dfrac{\mathbf{1}}{\mathbf{rank}}$} of the first relevant result.
  \vspace{0.2cm}
  \begin{center}
    \begin{tabular}{lll}
      \toprule
      Query & First relevant result & Reciprocal rank \\
      \midrule
      A & Rank 1 & 1 \\
      B & Rank 2 & 0.5 \\
      C & Missing from top 20 & 0 \\
      \bottomrule
    \end{tabular}
  \end{center}
  \begin{block}{Illustrative MRR@20}
    $(1 + 0.5 + 0) / 3 = 0.50$\\[0.1cm]
    Higher is better; use the same queries, relevance labels, and cutoff.
  \end{block}
\end{frame}

\begin{frame}{Validation: Backtesting}
  \begin{enumerate}
    \item Take historical customer search queries.
    \item Run each query against PostgreSQL and Elasticsearch.
    \item Share both ordered lists of 20 results with domain experts.
    \item Ask experts to vote for the list they would expect to see.
  \end{enumerate}
  \begin{block}{What the votes measure}
    Expert preference between result lists, not customer funnel selections. List-level votes are not MRR relevance labels.
  \end{block}
\end{frame}
