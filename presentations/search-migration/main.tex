\section{Problem \& Goal}
\subsection{Why and how this became an initiative}

\begin{frame}{Problem \& Goal: Initiative}
  \hypertarget{sec:problem}{}
  \begin{itemize}
    \item The current search worked and users \textbf{trusted} its \textbf{fuzzy matching} behavior.
    \item It ran on PostgreSQL with \textbf{16 CPU} cores and \textbf{64 GB} memory.
    \item At around \textbf{10M} records, \textbf{p99} latency was about \textbf{10 seconds}.
    \item Maximum sustainable \textbf{throughput} was around \textbf{2 requests per second}.
  \end{itemize}

  \vspace{0.5cm}
  \begin{block}{Goal}
    Bring \textbf{p99} search latency \textbf{under 1 second}, increase \textbf{throughput} by at least \textbf{10x}, and \textbf{preserve the ranking behavior} users already trusted.
  \end{block}
\end{frame}

\begin{frame}{Problem \& Goal: Options}
  After applying the available PostgreSQL and index optimizations, the remaining options were:
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.48\textwidth}
      \begin{itemize}
        \item Scale up the database.
          \begin{itemize}
            \item More CPU helped throughput and concurrency, but not single-query latency.
            \item It also kept search coupled to an expensive source-of-truth database.
          \end{itemize}
        \item Move search to a dedicated system.
      \end{itemize}

    \column{0.48\textwidth}
      \begin{block}{TBD: scaling benchmark}
        \small
        \begin{tabular}{@{}ll@{}}
          Current p99 & \textcolor{WarmAccent}{\rule{3.2cm}{0.18cm}} 10s \\
          Target p99 & \textcolor{SecondaryTeal}{\rule{0.35cm}{0.18cm}} <1s \\
          Current RPS & \textcolor{WarmAccent}{\rule{0.45cm}{0.18cm}} 2 \\
          Target RPS & \textcolor{SecondaryTeal}{\rule{3.8cm}{0.18cm}} $\geq$20
        \end{tabular}
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Problem \& Goal: Tool Investigation}
  We evaluated dedicated search engines against the requirements that mattered for production.

  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.5\textwidth}
      \begin{itemize}
        \item Functional and non-functional fit.
        \item Strong operational community.
        \item Easy to evolve as search behavior changes.
        \item Managed and self-hosted deployment options.
      \end{itemize}

    \column{0.44\textwidth}

        \begin{itemize}

\section{Search Concepts}
\subsection{Text similarity vs search}

\begin{frame}{Search Concepts: Why This Matters}
  \hypertarget{sec:search-concepts}{}
  Because PostgreSQL \texttt{pg\_trgm} and Elasticsearch BM25 match and rank text differently, we need a few concepts before discussing the migration.

  \vspace{0.25cm}
  \begin{itemize}
    \item Trigram similarity: how the old search matched fuzzy text.
    \item Trigram indexes: why p99 latency still had a tail cost.
    \item BM25: how Elasticsearch scores analyzed search terms.
    \item Matching tradeoffs and query fuzziness: where behavior changes.
  \end{itemize}
\end{frame}

\begin{frame}{Search Concepts: Trigram Similarity}
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.52\textwidth}
      \begin{itemize}
        \item It compares text by slicing strings into 3-character fragments.
        \item Similar strings share many trigrams.
        \item Good for typos, spacing differences, and fuzzy names.
        \item Language-agnostic character matching; no language-specific stemmer required.
      \end{itemize}

      \vspace{0.05cm}
      \begin{exampleblock}{Example}
        \texttt{search} $\rightarrow$ \texttt{\{ s, se, sea, ear, arc, rch, ch \}}\\
        \texttt{serch} $\rightarrow$ \texttt{\{ s, se, ser, erc, rch, ch \}}
      \end{exampleblock}

    \column{0.42\textwidth}
      \begin{block}{Shared fragments}
        \centering\Large 4
      \end{block}

      \begin{block}{Unique fragments}
        \centering\Large 9
      \end{block}

      \vspace{0.2cm}
      \begin{center}
        \large similarity = 4 / 9 = 0.44
      \end{center}
  \end{columns}
\end{frame}

\begin{frame}{Search Concepts: Trigram Indexes}
  \footnotesize
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.5\textwidth}
      \begin{block}{Bounded-field work model}
        \begin{itemize}
          \item Query trigram extraction is cheap and bounded.
          \item For names under 50 chars, each recheck is effectively constant.
          \item Approx order: $O(I + C)$ for fixed-size names.
          \item Top-20 ranking still examines the candidates.
        \end{itemize}
      \end{block}

      \begin{itemize}
        \item $I$: index pages/postings touched
        \item $C$: candidates after index filtering
      \end{itemize}

    \column{0.46\textwidth}
      \begin{block}{Example query}
        \texttt{WHERE name \% :q}\\
        \texttt{ORDER BY similarity(name, :q) DESC}\\
        \texttt{LIMIT 20}
      \end{block}

      \begin{block}{What changes the cost}
        \begin{itemize}
          \item Common trigrams touch more postings.
          \item Lower threshold increases $C$ and recall.
          \item Higher threshold reduces $C$ and tail work.
        \end{itemize}
      \end{block}

      \begin{block}{p99 impact}
        Sorting 20 rows is cheap; finding the top 20 among many candidates is not.
      \end{block}
  \end{columns}
\end{frame}
\begin{frame}{Search Concepts: BM25 Formula}
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.44\textwidth}
      \begin{block}{Analyzer creates terms}
        \texttt{apple apple banana}\\[0.2cm]
        $\rightarrow$ \texttt{apple}, \texttt{apple}, \texttt{banana}
      \end{block}

      \begin{block}{Base idea}
        \centering
        $\text{score} \approx tf \times idf$
      \end{block}

      \begin{itemize}
        \item $tf$: term frequency in this document
        \item $idf$: rarity across all documents
      \end{itemize}

    \column{0.52\textwidth}
      \begin{block}{BM25 extension}
        \scriptsize
        \[
          \sum_{t \in q} \operatorname{idf}(t) \cdot
          \frac{tf(t,d)(k_1 + 1)}
          {tf(t,d) + k_1\left(1 - b + b\frac{|d|}{avgdl}\right)}
        \]
      \end{block}

      \begin{itemize}
        \item Rare terms still matter most.
        \item Repeated terms help, but the gain saturates.
        \item Long noisy documents are normalized against average length.
      \end{itemize}
  \end{columns}

  \vspace{0.05cm}
  \begin{block}{Key distinction}
    \texttt{pg\_trgm} is text similarity. BM25 scores analyzed search terms.
  \end{block}
\end{frame}

\begin{frame}{Search Concepts: Matching Tradeoffs}
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.48\textwidth}
      \begin{block}{Similarity wins}
        Query: \texttt{jon smyth}\\[0.15cm]
        Record: \texttt{john smith}
      \end{block}

      \begin{itemize}
        \item Same name intent, different spelling.
        \item Shared character fragments keep the match close.
        \item BM25 may miss or weaken it because exact terms differ.
      \end{itemize}

    \column{0.48\textwidth}
      \begin{block}{BM25 wins}
        Query: \texttt{senior backend engineer}\\[0.15cm]
        Record: \texttt{backend engineer with search experience}
        \\[0.15cm]{\scriptsize With simple word / space tokenization}
      \end{block}

      \begin{itemize}
        \item Space-separated terms contribute to relevance.
        \item Rare terms carry more signal than common ones.
        \item Similarity alone treats this as broad text overlap.
      \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Search Concepts: Query Fuzziness}
  \small
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.48\textwidth}
      \begin{block}{Query shape}
        \texttt{match} analyzes text into terms and BM25 scores the matching documents.
      \end{block}

      \begin{itemize}
        \item Exact filters cut the candidate set first.
        \item Text clauses score the remaining candidates.
        \item Boosts decide which fields matter more.
      \end{itemize}

    \column{0.48\textwidth}
      \begin{block}{Fuzziness}
        \texttt{fuzziness: AUTO} expands terms to nearby spellings.
      \end{block}

      \begin{itemize}
        \item Better recall for typos and name variants.
        \item More expanded terms means more work per query.
        \item Prefix length and max expansions control latency.
      \end{itemize}
  \end{columns}

  \vspace{0.05cm}
  \begin{block}{Latency tradeoff}
    Target fuzziness at the fields that need it; broad fuzzy matching can increase p99 even when average latency looks fine.
  \end{block}
\end{frame}




\section{Migration Design}
\subsection{Moving data without overloading PostgreSQL}

\begin{frame}{Migration Design: Requirements}
  \hypertarget{sec:migration-design}{}
  \begin{itemize}
    \item Deploy Elasticsearch quickly with low operational overhead.
    \item Replicate required PostgreSQL data into Elasticsearch.
    \item Preserve fuzzy, language-agnostic name search behavior.
    \item Benchmark old and new ranking behavior.
    \item Prove the new path before production rollout.

\begin{frame}{Migration Design: Elasticsearch Cluster}
  \begin{itemize}
    \item \textbf{Cluster baseline:} 3 data nodes, each 4 cores / 8 GB RAM.
    \item \textbf{Index settings:} 3 primary shards and 1 replica.
    \item \textbf{Allocation:} Elasticsearch places primaries and replicas; we only configure shard and replica counts.
    \item \textbf{Read/write behavior:} reads can use primary or replica copies; writes go through the primary and replicate.
    \item \textbf{Shard sizing:} start around 20--50 GB per primary shard, then validate with real data and p95/p99 latency.
    \item \textbf{Managed cluster caveat:} CPU credits can distort short benchmarks; track credit balance with p95/p99 latency.
  \end{itemize}
\end{frame}

\begin{frame}{Migration Design: Document Identity}
  \begin{itemize}
    \item Elasticsearch document ID reused the PostgreSQL row ID.
    \item Engineers could debug the same entity across systems.
    \item Retries became idempotent updates, not duplicate documents.
    \item Replay became a normal recovery path.
  \end{itemize}
\end{frame}

\section{Index Design}
\subsection{Recreating fuzzy-name signals in Elasticsearch}

\begin{frame}{Index Design: Why These Views}
  \footnotesize
  \begin{block}{Selection rule}
    Each field view was chosen to recover one search behavior from the old system, then kept only if it improved ranking without hurting p95/p99 too much.
  \end{block}

  \vspace{0.05cm}
  \begin{tabular}{lll}
    \toprule
    Search need & Field view & Why it landed \\
    \midrule
    Normal name words & Standard text & BM25 baseline \\
    Exact company name & Normalized keyword & Strong cheap boost \\
    Partial input & Prefix & Early keystrokes \\
    Typos / OCR fragments & Character n-gram & Trigram-like recall \\
    Ordered multi-word names & Word shingle & Adjacent-word signal \\
    \bottomrule
  \end{tabular}

  \vspace{0.18cm}
  \begin{exampleblock}{Why word shingle}
    \texttt{progema fastighets} becomes its own adjacent-token signal, so it can outrank loose matches that contain only \texttt{progema} or \texttt{fastighets} separately.
  \end{exampleblock}
\end{frame}

\begin{frame}{Index Design: Field Views}
  \hypertarget{sec:index-design}{}
  \scriptsize
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.44\textwidth}
      \begin{tabular}{ll}
        \toprule
        Field view & Purpose \\
        \midrule
        Standard text & Full-text search \\
        Normalized keyword & Exact boost \\
        Prefix & Prefix match \\
        Character n-gram & Typo tolerance \\
        Word shingle & Multi-word boost \\
        \bottomrule
      \end{tabular}

    \column{0.52\textwidth}
      \begin{exampleblock}{Example: one source name}
        \texttt{Progema Fastighets AB}

        \vspace{0.10cm}
        \begin{tabular}{ll}
          \textbf{standard} & \texttt{progema}, \texttt{fastighets}, \texttt{ab} \\
          \textbf{keyword} & \texttt{progema fastighets ab} \\
          \textbf{prefix} & \texttt{p}, \texttt{pr}, \texttt{pro}, \texttt{prog...} \\
          \textbf{n-gram} & \texttt{pro}, \texttt{rog}, \texttt{oge...} \\
          \textbf{shingle} & \texttt{progema fastighets} \\
        \end{tabular}
      \end{exampleblock}

      \begin{block}{Query effect}
        One input can hit several boosted field views.
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Index Design: Analyzer Choices}
  \scriptsize
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.48\textwidth}
      \begin{exampleblock}{Field setting example}
        \begin{tabular}{ll}
          \textbf{field} & \textbf{index/search role} \\
          \hline
          \texttt{name} & main analyzed text \\
          \texttt{name.prefix} & edge-ngram for prefixes \\
          \texttt{name.ngram} & char n-gram for typos \\
          \texttt{name.raw} & normalized exact boost \\
          \texttt{search\_analyzer} & query-time tokenizer \\
        \end{tabular}
      \end{exampleblock}

      \begin{block}{Same text, different terms}
        \texttt{acme-fastighets.ab}\\[0.10cm]
        \textbf{standard-like:} \texttt{acme}, \texttt{fastighets}, \texttt{ab}\\
        \textbf{prefix field:} \texttt{a}, \texttt{ac}, \texttt{acm}, \texttt{acme...}
      \end{block}

    \column{0.48\textwidth}
      \begin{itemize}
        \item Multi-fields let one source field be indexed several ways.
        \item A query can send one user string and match several boosted fields.
        \item Each field can keep a lighter \texttt{search\_analyzer} at read time.
        \item More field views mean more index CPU, disk, refresh, and merge work.
        \item Custom analyzers can combine char filters, a tokenizer, and token filters.
        \item Use \texttt{\_analyze} on real names before committing the mapping.
      \end{itemize}

      \begin{block}{Design rule}
        Add a dedicated field when one tokenizer cannot serve exact, prefix, typo-tolerant, and ordered-name queries together.
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Index Design: Prefix Matching}
  \begin{exampleblock}{Input}
    \texttt{progema}
  \end{exampleblock}

  \vspace{0.05cm}
  \begin{center}
    \texttt{pro}, \texttt{prog}, \texttt{proge}, \texttt{progem}, \texttt{progema}
  \end{center}

  \vspace{0.45cm}
  \begin{block}{Use case}
    Users often type only the beginning of a name and expect autocomplete-like behavior.
  \end{block}
\end{frame}

\begin{frame}{Index Design: Character N-Grams}
  \begin{exampleblock}{Input}
    \texttt{progema}
  \end{exampleblock}

  \vspace{0.05cm}
  \begin{center}
    \texttt{pro}, \texttt{rog}, \texttt{oge}, \texttt{gem}, \texttt{ema}
  \end{center}

  \vspace{0.45cm}
  \begin{block}{Use case}
    This was the closest Elasticsearch signal to the old \texttt{pg\_trgm} behavior.
  \end{block}
\end{frame}

\begin{frame}{Index Design: Word Shingles}
  \footnotesize
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.47\textwidth}
      \begin{exampleblock}{Input}
        \texttt{progema fastighets ab}
      \end{exampleblock}

      \vspace{-0.05cm}
      \begin{center}
        \texttt{progema}\\
        \texttt{fastighets}\\
        \texttt{ab}\\[0.04cm]
        \texttt{progema fastighets}\\
        \texttt{fastighets ab}\\
        \texttt{progema fastighets ab}
      \end{center}

    \column{0.49\textwidth}
      \begin{block}{What it adds}
        Adjacent word groups become searchable tokens.
      \end{block}

      \begin{itemize}
        \item Preserves order and nearby-word context.
        \item Boosts exact company-name fragments.
        \item Separates phrase matches from loose word overlap.
        \item Complements n-grams; not typo tolerance.
      \end{itemize}

      \begin{block}{Ranking effect}
        \texttt{progema fastighets} prefers the full company name over a one-word overlap.
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Index Design: Size And Query Cost}
  \footnotesize
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.48\textwidth}
      \begin{tabular}{ll}
        \toprule
        Field view & Cost shape \\
        \midrule
        Exact keyword & Small \\
        Prefix & Medium \\
        Character n-gram & High postings count \\
        Word shingle & More phrase terms \\
        Reordered name field & Extra full field \\
        \bottomrule
      \end{tabular}

      \vspace{0.24cm}
      \begin{block}{Estimate before full rollout}
        Index a 1M-row sample, inspect \texttt{\_disk\_usage}, term counts from \texttt{\_analyze}, and query profiles. Multiply by expected corpus size with replica count included.
      \end{block}

    \column{0.48\textwidth}
      \begin{block}{How it helps performance}
        Precomputed fields move work from query time to indexing time.
      \end{block}

      \begin{itemize}
        \item Prefix fields avoid expensive broad fuzzy-prefix queries.
        \item N-grams recover typo tolerance without scanning every candidate.
        \item Shingles boost ordered fragments with fewer phrase-query surprises.
        \item Exact/normalized fields give cheap high-confidence boosts.
        \item Keep min/max grams tight and drop fields that do not move ranking metrics.
      \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Index Design: Query Strategy}
  \begin{itemize}
    \item Combine several name signals at query time.
    \item Boost exact and normalized matches.
    \item Use prefix fields for partial input.
    \item Use n-gram fields for misspellings and fuzzy fragments.
    \item Use shingles to preserve ordered multi-word context.
    \item Query reordered name fields for last-first / first-last variants.
  \end{itemize}

  \vspace{0.45cm}
  \begin{block}{Outcome}
    Around 90\% similar ranking experience with latency up to about 100 ms.
  \end{block}
\end{frame}

\section{Validation}
\subsection{Proving relevance before production rollout}

\begin{frame}{Validation: Backtesting}
  \hypertarget{sec:validation}{}
  \footnotesize
  \begin{itemize}
    \item Use historical user queries.
    \item Compare against results users actually selected.
    \item Measure whether Elasticsearch returned the same expected result.
    \item Track ranking differences with metrics such as MRR.
  \end{itemize}

  \vspace{0.25cm}
  \begin{block}{Why MRR fits ranking validation}
    Mean Reciprocal Rank rewards putting the first correct result near the top. That matches this search space: users usually need one right company quickly, and moving it from rank 8 to rank 2 matters much more than small changes lower in the list.
  \end{block}
\end{frame}

\begin{frame}{Validation: Generated Cases}
  \begin{itemize}
    \item Generate OCR-like errors.
    \item Add human typos and missing characters.
    \item Change spacing and casing.
    \item Remove words or groups of words.
    \item Test clean names and messy real-world inputs.
  \end{itemize}
\end{frame}

\begin{frame}{Validation: Human Judgement}
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.42\textwidth}
      \begin{itemize}
        \item Run the same query against both systems.
        \item Ask domain experts which result looked better.
        \item Treat \texttt{both} as acceptable ranking parity.
      \end{itemize}

      \vspace{0.25cm}
      \begin{block}{Decision signal}
        Elasticsearch was preferred or tied in 80\% of reviewed cases.
      \end{block}

\section{Outcome}
\subsection{Latency gains, trade-offs, and lessons}

\begin{frame}{Outcome: Before And After}

\begin{frame}{Outcome: Lessons}
  \begin{itemize}
    \item Search migration is not only a database replacement.
    \item The replication path needs observability and replayability.
    \item Analyzer design matters as much as the search engine.
    \item Ranking metrics are useful, but human judgement can change the decision.
    \item Idempotent writes and checkpoints reduce operational risk.
  \end{itemize}
\end{frame}

\begin{frame}{Outcome: Closing}
  \begin{center}
    \Large
    Elasticsearch did not replace \texttt{pg\_trgm} directly.\\[0.35cm]
    It recreated the useful signals through analyzers, weighting, benchmarks, and a replayable indexing pipeline.
  \end{center}
